# ============================================================
#  ДАТИРОВАННЫЕ СОБЫТИЯ ПРОТЕКТОРАТА — дизайн-док [MOR], Шаг 9
#
#  Цепочки 1936-1937 годов: «настоящее протектората», речь Лекаша в
#  еврейской общине, «Красный июнь», фосфатный кризис, рождение
#  независимой компартии, I Конгресс Марокканского комитета действий и
#  ноябрьский политический кризис, неурожай 1937 года.
#
#  СОГЛАШЕНИЯ ФАЙЛА
#  * Все события требуют, чтобы Марокко всё ещё было протекторатом, то
#    есть пока у него висит шкала MOR_power_balance (Шаг 7 снимает её при
#    независимости и приходе коммунистов). Цепочка сама гаснет, если
#    Марокко вышло из-под протектората.
#  * «Влияние генерал-резидента N %» = шкала MOR_power_balance уходит на
#    N десятых в соответствующую сторону (минус — Мухаммед V, плюс —
#    резидент), как и в решениях Шага 7.
#  * «Очки автономии» = add_autonomy_score (Марокко — субъект Франции).
#  * «НД» = национальные духи common/ideas/MOR june crisis ideas (Morocco).txt.
#  * «Выполнение текущего фокуса будет прерван, начнётся фокус „Провести
#    трудовые реформы“, и его нельзя будет прервать»: движок не умеет ни
#    прерывать, ни начинать фокус эффектом — единственная гарантированная
#    форма этой договорённости = complete_national_focus. Фокус завершается
#    сразу, поэтому его и невозможно прервать; перед этим ставится
#    mor_labor_reforms_forced_flag, чтобы Шаг 8 отличал «навязанные»
#    реформы от добровольных.
#  * Отложенные события («скрытый эффект» доки) спрятаны в hidden_effect,
#    поэтому в тултипе варианта их не видно.
#
#  ФЛАГИ СОБЫТИЙ (для будущих шагов):
#    mor_protectorate_intro_flag        сыграно событие о настоящем протектората
#    mor_jewish_speech_flag             речь Лекаша в Уджде
#    mor_jewish_communists_flag         фельетон о «красных соколах»
#    mor_red_june_happened_flag         Красный июнь состоялся (любая ветка)
#    mor_red_june_crackdown_flag        выбрана силовая развязка
#    mor_labor_reforms_forced_flag      трудовые реформы навязаны улицей
#    mor_phosphate_concessions_flag     уступки на рудниках Хурибги
#    mor_legion_at_houribgha_flag       по рудникам ударил Иностранный легион
#    mor_mcp_independent_flag           Партия коммунистов Марокко отделилась от ФКП
#    mor_red_june_apogee_flag           восстание достигло апогея
#    mor_mac_congress_flag              I Конгресс МКД состоялся
#    mor_mac_crisis_pending_flag        ноябрьский кризис идёт, ответ не дан
#    mor_mac_crisis_waiting_flag        Марокко «тянет с ответом» (см. on_actions)
#    mor_mac_crisis_answered_flag       ответ националистам дан
#    mor_mac_arrests_flag               лидеры МКД арестованы и отпущены
#    mor_mac_negotiated_flag            МКД приглашён к столу переговоров
#    mor_harvest_failure_flag           неурожайный год наступил
#  * Каждый ивент лежит в блоке country_event = { id = NW_MOR.N ... }. Без
#    обёртки и ключа id движок ивент не регистриует вовсе: в error.log это
#    "Unexpected token: NW_MOR.N", а в setup.log — "Events loaded ... #0".
#  * Поле icon у ивента невалидно ("Unexpected token: icon"), поэтому картинка
#    одна — picture =, и спрайты взяты из NWR (gfx/event_pictures).
#  * Дат у ивентов нет: date = внутри ивента не работает, а автоматические
#    ивенты движок перепроверяет раз в ~20 дней. Дни берёт из дизайн-дока
#    common/on_actions/MOR on_actions date (Morocco).txt, поэтому все ивенты
#    is_triggered_only = yes, fire_only_once снят, а однократность держится
#    на флагах NW_MOR_N_fired_flag.
#  * Каждый ивент помечает себя в логах строкой immediate = { log = ... }.

# ============================================================

# --- 4. Красный июнь в Касабланке — 12 июня 1936 ---
# Событие приходит, если к 11 июня фокус «Провести трудовые реформы» так
# и не завершён: волнение перекидывается с метрополии само.
country_event = {
	id = NW_MOR.4
	picture = GFX_report_event_worker_protests
	title = NW_MOR.4.t
	desc = NW_MOR.4.d
	is_triggered_only = yes

	immediate = { log = "[GetDateText]: [Root.GetName] ([Root.GetTag]): event NW_MOR.4" }

	# Пойти на уступки и договориться с лидерами.
	option = {
		name = NW_MOR.4.a
		set_country_flag = mor_red_june_happened_flag
		if = {
			limit = {
				NOT = { has_completed_focus = MOR_Conduct_labor_reforms }
			}
			set_country_flag = mor_labor_reforms_forced_flag
			complete_national_focus = MOR_Conduct_labor_reforms
		}
		add_autonomy_score = {
			value = 50
			localization = mor_autonomy_reason_red_june
		}
		add_power_balance_value = {
			id = MOR_power_balance
			value = -0.1
			tooltip_side = MOR_sultan_side
		}
		add_timed_idea = {
			idea = mor_aftermath_of_red_june_light
			days = 180
		}
		ai_chance = {
			base = 60
		}
	}

	# Использовать колониальные войска для разгона бунтовщиков!
	option = {
		name = NW_MOR.4.b
		set_country_flag = mor_red_june_happened_flag
		set_country_flag = mor_red_june_crackdown_flag
		add_stability = 0.05
		add_political_power = -50
		add_power_balance_value = {
			id = MOR_power_balance
			value = 0.1
			tooltip_side = MOR_resident_side
		}
		add_idea = mor_workers_unrest_in_casablanca
		hidden_effect = {
			log = "[GetDateText]: MOR NW_MOR.4 -> разгон забастовки, через 2 дня фосфатный кризис"
			country_event = { id = NW_MOR.5 days = 2 }
		}
		ai_chance = {
			base = 40
		}
	}
}

# --- 10. Начало ноябрьского политического кризиса ---
# Три ответа, и вид ответа зависит от того, какой из фокусов по МКД уже
# пройден. Если не пройден ни один — «тянем с ответом», и событие-последствие
# приходит позже, когда фокус всё же завершат (common/on_actions/
# MOR power balance on actions (Morocco).txt).
country_event = {
	id = NW_MOR.10
	picture = GFX_report_event_generic_rally_3
	title = NW_MOR.10.t
	desc = NW_MOR.10.d
	is_triggered_only = yes

	immediate = { log = "[GetDateText]: [Root.GetName] ([Root.GetTag]): event NW_MOR.10" }

	# Султан не хочет ввязываться в это, генерал-резидент разберётся сам.
	option = {
		name = NW_MOR.10.a
		trigger = {
			has_completed_focus = MOR_Oppose_the_Moroccan_Action_Committee
		}
		set_country_flag = mor_mac_crisis_answered_flag
		clr_country_flag = mor_mac_crisis_waiting_flag
		hidden_effect = {
			log = "[GetDateText]: MOR NW_MOR.10 -> резидент разбирается сам, через 21 день аресты лидеров МКД"
			country_event = { id = NW_MOR.11 days = 21 }
		}
		ai_chance = {
			base = 1
		}
	}

	# Султан симпатизирует им, и лично вызвался провести переговоры.
	option = {
		name = NW_MOR.10.b
		trigger = {
			has_completed_focus = MOR_Support_the_Moroccan_Action_Committee
		}
		set_country_flag = mor_mac_crisis_answered_flag
		clr_country_flag = mor_mac_crisis_waiting_flag
		hidden_effect = {
			log = "[GetDateText]: MOR NW_MOR.10 -> султан берёт переговоры на себя, через 7 дней приглашение МКД"
			country_event = { id = NW_MOR.12 days = 7 }
		}
		ai_chance = {
			base = 1
		}
	}

	# Потянем с ответом.
	option = {
		name = NW_MOR.10.c
		trigger = {
			NOT = { has_completed_focus = MOR_Oppose_the_Moroccan_Action_Committee }
			NOT = { has_completed_focus = MOR_Support_the_Moroccan_Action_Committee }
		}
		set_country_flag = mor_mac_crisis_waiting_flag
		ai_chance = {
			base = 1
		}
	}
}
