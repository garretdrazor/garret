# ============================================================
#  ДАТЫ ЦЕПОЧКИ ПРОТЕКТОРАТА — дизайн-док [MOR], Шаг 9
#
#  Датированные события нельзя зашивать в сам ивент полем date =: поле
#  невалидно (error.log: "Unexpected token: date"), а автоматические события
#  движок перепроверяет раз в ~20 дней, поэтому окно «дата в дату» всё равно
#  не срабатывает. Все события цепочки сделаны is_triggered_only = yes и
#  вызываются отсюда.
#
#  on_daily_MOR — страновой хук (область выполнения — Марокко), он и задаёт
#  точный день: date > <канун нужного дня> выполняется ровно с нужной даты.
#  Окно примерно в пять недель страхует случай, когда в нужный день Марокко
#  ещё не под протекторатом либо событие перечеркнуто пройденным фокусом;
#  для поздних стартов все окна уже прошли, и цепочка молчит.
#
#  Флаги mor_dated_N_fired_flag отвечают за однократность вместо снятого
#  fire_only_once. Цепочка живёт, пока у Марокко висит шкала MOR_power_balance.
#
#  ДАТЫ ИЗ ДОКА:
#    16.02.1936  mor_dated.2   выступления в еврейской общине
#    02.06.1936  mor_dated.3   вопиющее поведение еврейских коммунистов
#    12.06.1936  mor_dated.4   Красный июнь в Касабланке
#    25.10.1936  mor_dated.9   I Конгресс Марокканского комитета действий
#    03.02.1937  mor_dated.13  неурожайный год
#  Остальные события цепочки (5-8, 10-12) приходят по цепочке из предыдущих
#  событий и из common/on_actions/MOR power balance on actions (Morocco).txt.
# ============================================================

on_actions = {

	on_startup = {
		effect = {
			if = {
				limit = { has_start_date < 1936.01.02 }
				MOR = {
					country_event = { id = mor_dated.1 hours = 0 }
					country_event = { id = mor_dated.2 days = 46 }
					country_event = { id = mor_dated.3 days = 153 }
					country_event = { id = mor_dated.4 days = 163 }
					country_event = { id = mor_dated.9 days = 298 }
					country_event = { id = mor_dated.13 days = 399 }
				}
			}
		}
	}

}
